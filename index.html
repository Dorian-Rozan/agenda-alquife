<script>

    // Variables globales

    let currentAppointment = {};

    let appointments = JSON.parse(localStorage.getItem('appointments')) || [];

    let selectedIdToCancel = null;

 

    // Horas bloqueadas por defecto (Ej: comida 14:00 - 15:00) - Modificable

    const blockedHours = ["14:00", "14:15", "14:30", "14:45"];

 

    // Configurar fecha mínima de hoy en el calendario al cargar

    document.addEventListener('DOMContentLoaded', () => {

        const today = new Date().toISOString().split('T')[0];

        document.getElementById('date').setAttribute('min', today);

    });

 

    function showScreen(screenId) {

        document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));

        document.getElementById(screenId).classList.add('active');

    }

 

    function generateTimeSlots() {

        const dateInput = document.getElementById('date').value;

        if (!dateInput) return;

        

        const timeSelect = document.getElementById('time');

        timeSelect.innerHTML = '';

        

        // Generar horas de 09:00 a 19:30 en intervalos de 15 min

        let startTime = 9 * 60; // 9:00 en minutos

        let endTime = 19 * 60 + 30; // 19:30 en minutos

 

        for (let time = startTime; time <= endTime; time += 15) {

            let hours = Math.floor(time / 60);

            let mins = time % 60;

            let timeString = `${hours.toString().padStart(2, '0')}:${mins.toString().padStart(2, '0')}`;

            

            // Verificar si la hora está bloqueada o ya reservada ese día

            let isBlocked = blockedHours.includes(timeString);

            let isBooked = appointments.some(app => app.date === dateInput && app.time === timeString);

 

            if (!isBlocked && !isBooked) {

                let option = document.createElement('option');

                option.value = timeString;

                option.textContent = timeString;

                timeSelect.appendChild(option);

            }

        }

        document.getElementById('time-container').style.display = 'block';

    }

 

    function getDayOfWeek(dateString) {

        const days = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado'];

        const d = new Date(dateString);

        return days[d.getDay()];

    }

 

    function goToConfirmation() {

        const name = document.getElementById('name').value.trim();

        const phone = document.getElementById('phone').value.trim();

        const date = document.getElementById('date').value;

        const time = document.getElementById('time').value;

 

        if (!name || !date || !time) {

            alert("Por favor, rellene el nombre, fecha y hora.");

            return;

        }

 

        const dayName = getDayOfWeek(date);

        const formattedDate = date.split('-').reverse().join('/');

 

        currentAppointment = {

            id: Date.now().toString(),

            name, phone, date, time, dayName, formattedDate

        };

 

        document.getElementById('conf-name').textContent = name;

        document.getElementById('conf-phone').textContent = phone || "No proporcionado";

        document.getElementById('conf-date').textContent = `${dayName}, ${formattedDate}`;

        document.getElementById('conf-time').textContent = time;

 

        showScreen('screen-confirm');

    }

 

    function saveAppointment() {

        appointments.push(currentAppointment);

        localStorage.setItem('appointments', JSON.stringify(appointments));

        

        // Enviar mensaje de WhatsApp si hay teléfono

        if (currentAppointment.phone) {

            const msg = `Estimado cliente, a su petición le hemos asignado cita para visitar el Estudio que sita en CL Sierra de Alquife el ${currentAppointment.dayName}, ${currentAppointment.formattedDate} a las ${currentAppointment.time}, por lo queda reservada. Le mostrará el estudio el Agente Rubén que le estará esperando en la puerta del Centro de Salud que se encuentra en el mismo parque de la CL Sierra de Alquife. Rogamos puntualidad en la medida de lo posible, si desean cancelar la visita, les rogamos nos llame o escriban SMS o Text a WhatsApp diciendo el nombre a quien está la visita día y hora y la palabra ANULAR. Gracias por su colaboración.`;

            

            const encodedMsg = encodeURIComponent(msg);

            const cleanPhone = currentAppointment.phone.replace(/\D/g,''); // Elimina espacios y símbolos

            const waUrl = `https://wa.me/${cleanPhone}?text=${encodedMsg}`;

            window.open(waUrl, '_blank');

        }

 

        clearForm();

        showDashboard();

    }

 

    function clearForm() {

        document.getElementById('name').value = '';

        document.getElementById('phone').value = '';

        document.getElementById('date').value = '';

        document.getElementById('time-container').style.display = 'none';

        currentAppointment = {};

        showScreen('screen-booking');

    }

 

    function showDashboard() {

        renderAppointments();

        showScreen('screen-dashboard');

    }

 

    function renderAppointments() {

        const list = document.getElementById('appointments-list');

        const filter = document.getElementById('view-filter').value;

        list.innerHTML = '';

        

        const today = new Date();

        today.setHours(0,0,0,0);

 

        // Ordenar por fecha y hora

        let filtered = appointments.sort((a, b) => {

            return new Date(a.date + 'T' + a.time) - new Date(b.date + 'T' + b.time);

        });

 

        filtered.forEach(app => {

            const appDate = new Date(app.date);

            appDate.setHours(0,0,0,0);

            const diffTime = Math.abs(appDate - today);

            const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));

            

            let show = false;

            if (filter === 'all') show = true;

            else if (filter === 'today' && diffDays === 0 && appDate.getTime() === today.getTime()) show = true;

            else if (filter === '3days' && appDate >= today && diffDays <= 3) show = true;

            else if (filter === 'week' && appDate >= today && diffDays <= 7) show = true;

            else if (filter === 'month' && appDate >= today && diffDays <= 30) show = true;

 

            if (show) {

                const card = document.createElement('div');

                card.className = 'appointment-card';

                card.innerHTML = `<strong>${app.time}</strong> - ${app.dayName}, ${app.formattedDate}<br>

                                  Cliente: ${app.name} <br> Tel: ${app.phone || 'N/A'}`;

                card.onclick = () => openDetails(app.id);

                list.appendChild(card);

            }

        });

 

        if (list.innerHTML === '') {

            list.innerHTML = '<p>No hay citas programadas para este periodo.</p>';

        }

    }

 

    function openDetails(id) {

        const app = appointments.find(a => a.id === id);

        if(!app) return;

        

        selectedIdToCancel = id;

        document.getElementById('det-name').textContent = app.name;

        document.getElementById('det-phone').textContent = app.phone || "N/A";

        document.getElementById('det-date').textContent = `${app.dayName}, ${app.formattedDate}`;

        document.getElementById('det-time').textContent = app.time;

        

        showScreen('screen-details');

    }

 

    function cancelAppointment() {

        if(confirm("¿Está seguro de que desea ANULAR esta cita?")) {

            appointments = appointments.filter(a => a.id !== selectedIdToCancel);

            localStorage.setItem('appointments', JSON.stringify(appointments));

            alert("Cita anulada con éxito.");

            showDashboard();

        }

    }
